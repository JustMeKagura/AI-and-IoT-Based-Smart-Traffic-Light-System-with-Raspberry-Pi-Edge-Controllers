# Install on the Pi with:
#   pip install -r requirements_edge.txt --break-system-packages

# Rules:

#     here for reproducibility.
#   - PyYAML for config.yaml parsing (shared/config_loader.py).
# =============================================================================

# Camera capture and JPEG encoding (headless = no GUI, safe on Pi)
opencv-python-headless>=4.9.0

# GPIO control via gpiozero (BCM pin numbering, clean LED abstraction)
gpiozero>=2.0

# GPIO backend for gpiozero on Raspberry Pi 5

lgpio>=0.2.0

# Config file parsing
